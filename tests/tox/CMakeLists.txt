cmake_minimum_required(VERSION 3.16)
project(c1_tox_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(C1_TOX_ROOT ../.. ABSOLUTE)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -Wno-misleading-indentation -Wno-missing-field-initializers")
include(${C1_TOX_ROOT}/tox/dependencies.cmake)
add_library(c1net STATIC ${C1_TOX_ROOT}/shared/net.cpp)
target_include_directories(c1net PUBLIC ${C1_TOX_ROOT}/shared ${C1_TOX_ROOT}/.build/include)
target_link_libraries(c1net PUBLIC pthread)
add_library(c1tox STATIC ${C1_TOX_ROOT}/tox/src/engine.cpp ${C1_TOX_ROOT}/tox/src/outbox.cpp ${C1_TOX_ROOT}/tox/src/files.cpp ${C1_TOX_ROOT}/tox/src/session.cpp)
target_include_directories(c1tox PUBLIC ${C1_TOX_ROOT}/tox/src ${C1_TOX_ROOT}/.deps/c-toxcore)
target_link_libraries(c1tox PUBLIC toxcore_static c1net pthread)
add_executable(tox-roundtrip roundtrip.cpp)
target_link_libraries(tox-roundtrip c1tox)
target_compile_options(tox-roundtrip PRIVATE -UNDEBUG)
set(LV_BUILD_CONF_PATH "${C1_TOX_ROOT}/shared/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(${C1_TOX_ROOT}/.deps/lvgl ${CMAKE_BINARY_DIR}/lvgl EXCLUDE_FROM_ALL)
target_compile_definitions(lvgl PUBLIC LV_USE_QRCODE=1)
include(${C1_TOX_ROOT}/tox/qr.cmake)
add_executable(tox-qr qr.cpp)
target_link_libraries(tox-qr c1toxqr c1tox)
target_compile_options(tox-qr PRIVATE -UNDEBUG)
add_library(c1toxmedia STATIC ${C1_TOX_ROOT}/tox/src/media.cpp ${C1_TOX_ROOT}/camera/src/album.cpp)
target_include_directories(c1toxmedia PRIVATE ${C1_TOX_ROOT}/.deps/lvgl/src/libs/gltf/stb_image)
target_link_libraries(c1toxmedia PUBLIC c1net)
add_executable(tox-preview ${C1_TOX_ROOT}/tox/src/main.cpp headless.cpp)
target_include_directories(tox-preview PRIVATE ${C1_TOX_ROOT}/.deps/lvgl/src/libs/tiny_ttf)
target_link_libraries(tox-preview c1tox c1toxqr c1toxmedia lvgl m)
add_executable(tox-network-probe network_probe.cpp)
target_link_libraries(tox-network-probe c1tox)
add_executable(tox-qr-image qr_image.cpp)
target_link_libraries(tox-qr-image c1toxqr)
add_executable(tox-scan-test scan_image.cpp)
target_link_libraries(tox-scan-test c1toxqr)
target_compile_options(tox-scan-test PRIVATE -UNDEBUG)
add_executable(tox-session-test session.cpp)
target_link_libraries(tox-session-test c1tox)
target_compile_options(tox-session-test PRIVATE -UNDEBUG)
add_executable(tox-outbox-test outbox.cpp)
target_link_libraries(tox-outbox-test c1tox)
target_compile_options(tox-outbox-test PRIVATE -UNDEBUG)
add_executable(tox-media-preview-test media_preview.cpp)
target_link_libraries(tox-media-preview-test c1toxmedia)
target_include_directories(tox-media-preview-test PRIVATE ${C1_TOX_ROOT}/tox/src)
target_compile_options(tox-media-preview-test PRIVATE -UNDEBUG)
